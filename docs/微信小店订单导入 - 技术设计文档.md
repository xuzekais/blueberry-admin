## 一、技术架构

### 1.1 现有架构分析

项目采用前后端分离架构：
- **前端**: Vue 3 + Vite + Element Plus
- **后端**: Node.js + Express + MySQL
- **导入模块**: 基于策略模式的导入器架构

现有导入流程：
```
前端上传 Excel → /api/import/orders → importController.uploadOrders
  → importerManager.parseOrdersBySource(source, buffer)
  → miniProgramImporter.parseMiniProgramOrders(buffer)
  → 数据清洗/计算 → 写入数据库
```

### 1.2 扩展方案

**采用策略模式新增微信小店导入器**，最小化改动现有代码：

```javascript
// utils/importers/index.js
const importStrategies = {
    'miniprogram': miniProgramImporter.parseMiniProgramOrders,
    'wechat_shop': wechatShopImporter.parseWechatShopOrders,  // 新增
};
```

### 3.2 核心实现：wechatShopImporter.js

**文件路径**: `backend/utils/importers/wechatShopImporter.js`

```javascript
const xlsx = require('xlsx');

/**
 * 微信小店订单 Excel 导入解析器
 * @param {Buffer} buffer - Excel 文件 Buffer
 * @returns {Array} - 解析后的订单数据列表
 */
async function parseWechatShopOrders(buffer) {
    const workbook = xlsx.read(buffer, { type: 'buffer' });
    const sheetName = workbook.SheetNames[0];
    const sheet = workbook.Sheets[sheetName];
    
    // 解析为二维数组
    const rows = xlsx.utils.sheet_to_json(sheet, { header: 1, defval: '' });
    console.log(`微信小店订单：解析到 ${rows.length} 行数据，正在寻找表头...`);
    
    let headerRowIndex = -1;
    let headers = [];
    
    // 1. 寻找表头（根据微信小店导出格式的关键字段）
    for (let i = 0; i < rows.length; i++) {
        const row = rows[i];
        if (!row || row.length === 0) continue;
        
        const rowStr = row.join(',');
        // 微信小店表头特征字段
        if (rowStr.includes('订单编号') && rowStr.includes('收货人') && rowStr.includes('实付金额')) {
            headerRowIndex = i;
            headers = row.map(h => (h ? h.toString().trim() : ''));
            break;
        }
    }
    
    if (headerRowIndex === -1) {
        throw new Error('未找到有效的订单表头行，请检查文件格式是否为微信小店导出格式');
    }
    
    const parsedOrders = [];
    
    // 2. 解析数据行
    for (let i = headerRowIndex + 1; i < rows.length; i++) {
        const row = rows[i];
        if (!row || row.length === 0) continue;
        
        // 映射为对象
        const rowData = {};
        for (let j = 0; j < headers.length; j++) {
            rowData[headers[j]] = row[j] !== undefined ? row[j].toString().trim() : '';
        }
        
        // 3. 字段映射（PRD 3.2 字段映射规则）
        const orderNo = rowData['订单编号'];
        if (!orderNo) continue; // 跳过无订单号的行
        
        const receiverName = rowData['收货人'];
        const phone = rowData['联系电话'];
        const address = rowData['收货地址'];
        const totalPrice = parseFloat(rowData['实付金额'] || '0') || 0;
        const orderTime = rowData['下单时间'];
        const itemDesc = rowData['商家备注'] || ''; // 托寄物映射自商家备注
        const trackingNo = rowData['物流单号'] || '';
        const logisticsCompany = rowData['物流公司'] || '';
        
        // 4. 数据校验（PRD 3.4 异常处理规则）
        // 4.1 必填字段校验
        if (!receiverName) {
            console.warn(`第 ${i + 1} 行缺少收货人，已跳过`);
            continue;
        }
        if (!phone) {
            console.warn(`第 ${i + 1} 行缺少联系电话，已跳过`);
            continue;
        }
        if (!address) {
            console.warn(`第 ${i + 1} 行缺少收货地址，已跳过`);
            continue;
        }
        
        // 4.2 手机号格式校验
        const phoneRegex = /^1[3-9]\d{9}$/;
        if (!phoneRegex.test(phone.replace(/\s|-/g, ''))) {
            console.warn(`第 ${i + 1} 行手机号格式错误：${phone}，已跳过`);
            continue;
        }
        
        // 4.3 订单金额校验
        if (totalPrice < 0) {
            console.warn(`第 ${i + 1} 行订单金额为负数：${totalPrice}，已跳过`);
            continue;
        }
        
        // 5. 自动计算物流产品（PRD 3.2）
        const isGuangdong = address.startsWith('广东省') || address.includes('广东省');
        let expressProduct = logisticsCompany; // 优先使用 Excel 中的物流公司
        
        if (!expressProduct) {
            // 无值则自动判断
            expressProduct = isGuangdong ? '陆运包裹' : '顺丰特快';
        }
        
        // 6. 构建订单对象
        const mappedOrder = {
            original_order_no: orderNo,
            original_order_time: orderTime,
            original_tracking_no: trackingNo,
            original_logistics_company: logisticsCompany,
            
            // 映射目标字段
            receiver_name: receiverName,
            receiver_phone: phone,
            receiver_address: address,
            item_name: itemDesc, // 商家备注 → 托寄物
            total_price: totalPrice,
            express_product: expressProduct,
            
            // 渠道标识
            channel: 'wechat_shop'
        };
        
        parsedOrders.push(mappedOrder);
    }
    
    console.log(`微信小店订单：成功解析 ${parsedOrders.length} 条有效订单`);
    return parsedOrders;
}

module.exports = {
    parseWechatShopOrders
};
```

### 3.3 扩展导入策略管理器

**文件**: `backend/utils/importers/index.js`

```javascript
const miniProgramImporter = require('./miniProgramImporter');
const wechatShopImporter = require('./wechatShopImporter'); // 新增

const importStrategies = {
    'miniprogram': miniProgramImporter.parseMiniProgramOrders,
    'wechat_shop': wechatShopImporter.parseWechatShopOrders, // 新增
};

async function parseOrdersBySource(source, buffer) {
    const strategy = importStrategies[source];
    if (!strategy) {
        throw new Error(`未找到对于数据源 ${source} 的解析策略`);
    }
    return await strategy(buffer);
}

module.exports = {
    parseOrdersBySource
};
```

## 五、运费计算逻辑

**复用现有逻辑**（`importController.js` 中已实现）：

```javascript
// 判断是否广东
const isGuangdong = address.startsWith('广东省') || address.includes('广东省');

// 判断是否有红果/次果
let hasHongGuoOrCiGuo = false;
// ... 在解析 item_name 时判断

// 计算运费
let estimated_freight = 0;
if (isGuangdong) {
    estimated_freight = 10;
} else {
    if (hasHongGuoOrCiGuo) {
        estimated_freight = 0; // 红果/次果免运费
    } else {
        estimated_freight = 18;
    }
}
```

**注意**：微信小店导入时，`item_name` 来自"商家备注"字段，可能不包含果径信息。需要在导入器中增加果径识别逻辑，或者在导入后手动补充。
